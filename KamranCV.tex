% ---------- Commands ----------------------------------------------
% Name centered across the full page (label column + body)
\newcommand{\cvname}[1]{%
  \vspace*{-0.35in}%
  \noindent\hspace*{-\dimexpr\cvlabelwidth+\cvlabelgap\relax}%
  \makebox[\dimexpr\textwidth+\cvlabelwidth+\cvlabelgap\relax][c]{\Large\bfseries #1}%
  \par\vspace{14pt}}

% \cvsection{Line one\\Line two}  -> small-caps label in the left column.
% The label is a zero-height box, so whatever follows (text, a list,
% a \cvsubsection) starts on the same line as the label.
\newcommand{\cvsection}[1]{%
  \par\addvspace{16pt}\nointerlineskip
  \vbox to 0pt{\kern\baselineskip
    \hbox{\hskip-\dimexpr\cvlabelwidth+\cvlabelgap\relax
      \raisebox{0pt}[0pt][0pt]{\parbox[t]{\cvlabelwidth}{\raggedright\scshape #1}}}%
    \vss}%
  \prevdepth=0pt\nobreak}

% \cvsubsection{Title} -> bold-free sub-heading inside a section (e.g., "Keynotes")
% (no extra space when it directly follows \cvsection, so it sits beside the label)
\newcommand{\cvsubsection}[1]{\par
  \ifnum\lastpenalty=10000 \else\addvspace{8pt}\needspace{3\baselineskip}\fi
  \noindent #1\par\nobreak\addvspace{6pt}}

% \cventry{Bold title}{Dates}{Second line (department, institution)}
\newcommand{\cventry}[3]{%
  \noindent\textbf{#1}\hfill #2\par
  \ifx&#3&\else\noindent #3\par\fi
  \addvspace{10pt}}

% \cvdegree{Degree, Institution}{Dates}{Details (\\-separated)}
\newcommand{\cvdegree}[3]{%
  \noindent #1\hfill #2\par
  \ifx&#3&\else\vspace{4pt}{\leftskip=1.2em\noindent #3\par}\fi
  \addvspace{10pt}}

% \cvitem{Left text}{Right text (year)} -> one line with a right-aligned date
\newcommand{\cvitem}[2]{\noindent #1\hfill #2\par}

% Numbered list used for grants, publications, services, ...
\newlist{cvlist}{enumerate}{1}
\setlist[cvlist]{label=\arabic*.,leftmargin=*,align=right,
                 itemsep=2pt,topsep=2pt,parsep=0pt}
% Publication list numbered in reverse (newest first):
% \begin{publist}{J}{7}{7} -> [J7], [J6], ..., [J1]
\makeatletter
\protected\def\revpubnum{\number\numexpr\pubtotal+1-\c@enumi\relax}
\makeatother
\newenvironment{publist}[2]{%
  \def\pubtotal{#2}%
  \begin{enumerate}[label={[#1\revpubnum]},
                    leftmargin=2.6em,labelwidth=2.3em,labelsep=0.3em,align=left,
                    itemsep=3pt,topsep=2pt,parsep=0pt]}%
  {\end{enumerate}}

% Bullet list (research experience, talks, services)
\newlist{cvbullets}{itemize}{1}
\setlist[cvbullets]{label=\textbullet,leftmargin=1.2em,
                    itemsep=1pt,topsep=2pt,parsep=0pt}
